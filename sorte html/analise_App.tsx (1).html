async function uploadToDrive(token: string, blob: Blob, name: string) {
  const meta = JSON.stringify({ name, mimeType: "application/zip" });
  const form = new FormData();
  form.append("metadata", new Blob([meta], { type: "application/json" }));
  form.append("file", blob, name);
  const r = await fetch("https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart", {
    method: "POST", headers: { Authorization: `Bearer ${token}` }, body: form,
  });
  if (!r.ok) throw new Error(`Drive ${r.status}: ${(await r.json()).error?.message}`);
  const j = await r.json();
  return `https://drive.google.com/file/d/${j.id}/view`;
}

function downloadBlob(blob: Blob, name: string) {
  // Método 1: createObjectURL (desktop e maioria dos browsers)
  try {
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url; a.download = name; a.style.display = "none";
    document.body.appendChild(a);
    a.click();
    setTimeout(() => { try { document.body.removeChild(a); } catch {} URL.revokeObjectURL(url); }, 1500);
    return;
  } catch { /* tenta fallback */ }

  // Método 2: FileReader readAsDataURL (mobile iOS/Android)
  try {
    const reader = new FileReader();
    reader.onload = () => {
      const a = document.createElement("a");
      a.href = reader.result as string; a.download = name; a.style.display = "none";
      document.body.appendChild(a); a.click();
      setTimeout(() => { try { document.body.removeChild(a); } catch {} }, 1500);
    };
    reader.readAsDataURL(blob);
    return;
  } catch { /* tenta fallback */ }

  // Método 3: window.open (último recurso)
  try {
    const reader2 = new FileReader();
    reader2.onload = () => { window.open(reader2.result as string, "_blank"); };
    reader2.readAsDataURL(blob);
  } catch (e) {
    alert("Erro ao baixar. Use Chrome ou Firefox.\n" + String(e));
  }
}

        ctx.drawImage(img, 0, 0, size, size);

  async function neonQuery(sql: string, params: unknown[] = []): Promise<{ cols: string[]; rows: unknown[][] }> {
    const p = parseNeonConn(dbConnStr);
    if (!p) throw new Error("String de conexão inválida. Use: postgresql://user:pass@host/db");
    const res = await fetch(`https://${p.host}/sql`, {
      method: "POST",
      headers: { "Content-Type": "application/json", Authorization: `Bearer ${p.pass}` },
      body: JSON.stringify({ query: sql, params }),
    });
    if (!res.ok) {
      const err = await res.json().catch(() => ({}));
      throw new Error(err.message || `HTTP ${res.status}`);
    }
    const d = await res.json();
    // Neon HTTP returns { fields:[{name}], rows:[{col:val}] }
    const fields: string[] = (d.fields ?? []).map((f: { name: string }) => f.name);
    const rows: unknown[][] = (d.rows ?? []).map((r: Record<string, unknown>) => fields.map(f => r[f]));
    return { cols: fields, rows };
  }
